using Microsoft.Data.Sqlite;

namespace PlataformaStreaming
{
    internal class Program
    {
        static void Main(string[] args)
        {
            try
            {
                using (SqliteConnection conexion = ConexionDB.ObtenerConexion())
                {
                    // @ permite escribir uan cadena de texto en varias lineas sin tener que concatenarlas ni agregar \n manualmente (es de C#, no de SQL)

                    string sql = @"
                        
                        SELECT contenido_id, titulo, descripcion, anio_lanzamiento, tipo
                        FROM Contenidos
                        WHERE tipo = @tipo;

                    ";
                    // ContenidoGenero: su clave primaria se compone de dos campos para evitar asignar dos veces el mismo género al mismo contenido


                    using (SqliteCommand comando = new SqliteCommand(sql, conexion))
                    // creamos la variable comando para el objeto SqliteCommand que se conecta a conexion (que apunta a la DB) usando las instruciones de la variable sql (creada anteriormente como string sql) 
                    {
                        comando.Parameters.AddWithValue("@tipo", "Pelicula");

                        using (SqliteDataReader lector = comando.ExecuteReader()) // executeReader ejecuta la consulta y permite leer las filas devueltas
                        {
                            while (lector.Read()) // lector.Read avanza a la siguiente fila. Devuelve false cuando no quedan filas
                            {
                                Console.WriteLine("ID: " + lector["contenido_id"]); // lector[] obtiene el valor de la columna [] de la fila actual
                                Console.WriteLine("Título: " + lector["titulo"]);
                                Console.WriteLine("Descripcion: " + lector["descripcion"]);
                                Console.WriteLine("Año: " + lector["anio_lanzamiento"]);
                                Console.WriteLine("Tipo: " + lector["tipo"]);
                                Console.WriteLine("------------------------");
                            }
                        }


                            // comando.ExecuteNonQuery(); NO ES NECESARIO, solo se uasa para operaciones como INSERT, UPDATE, DELETE y creacion de tablas
                    }

                }

            }
            catch (Exception ex)
            {
                Console.WriteLine("Error: " + ex.Message);
            }

            Console.WriteLine("\nPresioná una tecla para salir...");
            Console.ReadKey();
        }
    }
}
